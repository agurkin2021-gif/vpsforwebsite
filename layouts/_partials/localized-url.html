{{- /* Resolve completed translations by source path; otherwise use the existing English page. */ -}} {{- $path := . -}} {{- $url := $path | relURL -}} {{- if or (strings.HasPrefix $path "#") (strings.HasPrefix $path "http") (strings.HasPrefix $path "mailto:") -}} {{- $url = $path -}} {{- else -}} {{- $parts := split $path "#" -}} {{- $lookupPath := index $parts 0 -}} {{- if in (slice "de" "it" "fa" "ar" "ja" "tr") site.Language.Lang -}} {{- /* Hugo menu URLs already include a project BaseURL path. GetPage expects the source path instead; strip only that known prefix. RelPermalink adds it back after choosing the German translation. */ -}} {{- $basePath := strings.TrimSuffix "/" (urls.Parse site.BaseURL).Path -}} {{- if and $basePath (strings.HasPrefix $lookupPath (printf "%s/" $basePath)) -}} {{- $lookupPath = strings.TrimPrefix $basePath $lookupPath -}} {{- else if and $basePath (eq $lookupPath $basePath) -}} {{- $lookupPath = "/" -}} {{- end -}} {{- end -}} {{- $target := site.GetPage $lookupPath -}} {{- if not $target -}} {{- $target = (index hugo.Sites 0).GetPage $lookupPath -}} {{- end -}} {{- with $target -}}{{- $url = .RelPermalink -}}{{- end -}} {{- if gt (len $parts) 1 -}}{{- $url = printf "%s#%s" $url (index $parts 1) -}}{{- end -}} {{- end -}} {{- return $url -}}
